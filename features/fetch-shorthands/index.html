$.get, $.post & $.getJSON

The one-liners. await $.getJSON('/api/users.json') and you have data; $.post() sends a form-encoded body; $.get() returns text or JSON based on the response.

Each shorthand returns the same Promise as $.ajax(). For $.get() and $.getJSON(), an object in the second argument becomes the query string; for $.post() it becomes an application/x-www-form-urlencoded body. A callback in the last position still works for older code.

JavaScript
let team = [];
const render = () => $('#fs-team').html(team.map(u =>
  '<div class="p"><b>' + $.escape(u.name) + '</b><small>' + $.escape(u.role) + ' · ' + u.commits + ' commits</small></div>').join(''));

$('#fs-load').on('click', async () => {
  team = await $.getJSON('/api/users.json');
  render();
  $.from('#fs-team .p', { y: 14, opacity: 0, duration: .3, stagger: .06 });
});

$('#fs-sort').on('click', () => {
  team.sort((a, b) => b.commits - a.commits);
  render();
});

$('#fs-quote').on('click', async () => {
  const n = 1 + Math.floor(Math.random() * 5);
  $('#fs-q').html(await $.get('/api/quotes/' + n + '.html'));
});
